    <script>
        const fpsDisplay = document.getElementById('fps-val');
        let frameTimes = [];

        function renderFPS(now) {
            while (frameTimes.length > 0 && frameTimes[0] <= now - 1000) {
                frameTimes.shift();
            }
            frameTimes.push(now);

            const currentFPS = frameTimes.length;
            fpsDisplay.innerText = currentFPS;

            if (currentFPS >= 50) {
                fpsDisplay.className = "text-cyan-400 font-bold";
            } else if (currentFPS >= 30) {
                fpsDisplay.className = "text-amber-400 font-bold";
            } else {
                fpsDisplay.className = "text-rose-500 font-bold";
            }

            requestAnimationFrame(renderFPS);
        }
        requestAnimationFrame(renderFPS);
    </script>
